"use client";

import { Button } from "@/components/ui/button";
import { useSession } from "next-auth/react";
import Link from "next/link";
import { useState, useEffect } from "react";
import { useRouter } from "next/navigation";

export default function HomePage() {

	}, [session]);

			setMousePosition({
				x: (e.clientX / window.innerWidth) * 100,
				y: (e.clientY / window.innerHeight) * 100,
			});
		};

		window.addEventListener("mousemove", handleMouseMove);
		return () => window.removeEventListener("mousemove", handleMouseMove);

				{[...Array(20)].map((_, i) => (
					<div
						key={i}
						className="absolute w-2 h-2 bg-white/10 rounded-full animate-pulse"
						style={{
							left: `${Math.random() * 100}%`,
							top: `${Math.random() * 100}%`,
							animationDelay: `${Math.random() * 2}s`,
							animationDuration: `${2 + Math.random() * 3}s`,
						}}
					></div>
				))}
			</div>

			<div className="absolute inset-0 bg-gradient-to-r from-transparent via-slate-900/50 to-transparent"></div>

					<div className="flex justify-between items-center">
						<div className="flex items-center gap-1.5 text-lg sm:text-2xl lg:text-3xl font-bold bg-gradient-to-r from-emerald-400 to-cyan-400 bg-clip-text text-transparent">

							<Button
								className="bg-violet-600 hover:bg-violet-700 text-xs sm:text-sm lg:text-base text-white shadow-lg hover:shadow-violet-500/25 px-2.5 py-1.5 sm:px-3 sm:py-2 lg:px-4"

							>
								<Link href="/feed">Feed</Link>
							</Button>
							<Button
								className="bg-emerald-600 hover:bg-emerald-700 text-xs sm:text-sm lg:text-base text-white shadow-lg hover:shadow-emerald-500/25 px-2.5 py-1.5 sm:px-3 sm:py-2 lg:px-4"
								asChild
							>
								<Link href="/entrar">Entrar</Link>
							</Button>

							<div

							>

										Monetize suas Respostas

								</p>

								<div className="mb-8 sm:mb-12 lg:mb-16 px-2">

										size="lg"
										className="group bg-gradient-to-r from-orange-500 via-red-500 to-pink-500 hover:from-orange-600 hover:via-red-600 hover:to-pink-600 text-white px-6 sm:px-8 lg:px-12 py-3 sm:py-4 lg:py-6 text-base sm:text-lg lg:text-xl font-semibold shadow-2xl hover:shadow-orange-500/50 transition-all duration-500 transform hover:scale-105 lg:hover:scale-110 relative overflow-hidden w-full sm:w-auto"

				</div>

					<div className="flex flex-col sm:flex-row justify-between items-center gap-4">
						<div className="text-xs sm:text-sm text-gray-500 text-center sm:text-left">
